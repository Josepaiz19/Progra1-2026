//Parte 1
#include <iostream>
using namespace std;
int main()
{
    float nota[5];
    float suma = 0;
    float promedio;
    float mayor, menor;
  

    for (int i = 0; i < 5; i++) {
        cout << "Ingrese nota " << i + 1<<":";
        cin >> nota[i];
        suma += nota[i];

        if (i == 0) {
            mayor = nota[i];
            menor = nota[i];
        }
        else{
            if (nota[i] > mayor) {
                mayor = nota[i];
            }
            if (nota[i] < menor) {
                menor = nota[i];
            }
        }
    }
    promedio = suma / 5;
    for (int i = 0; i < 5; i++) {
        cout << "Nota " << i + 1 <<"=" << nota[i] << endl;
    }
    cout << "El promedio es de: " << promedio << endl;
    cout << "La nota mayor es de: " << mayor << endl;
    cout << "La nota menor es de: " << menor << endl;
}
